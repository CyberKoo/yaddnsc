cmake_minimum_required(VERSION 3.28 FATAL_ERROR)
option(YADDNSC_CONSUMER_C_ONLY "Build with a pure C toolchain and no Glaze" OFF)
option(YADDNSC_CONSUMER_OLD_CXX_ABI "Compile crypto with the other libstdc++ ABI" OFF)
if(YADDNSC_CONSUMER_C_ONLY)
    project(yaddnsc_sdk_consumer LANGUAGES C)
    find_package(yaddnsc CONFIG REQUIRED COMPONENTS plugin_abi)
else()
    project(yaddnsc_sdk_consumer LANGUAGES C CXX)
    find_package(yaddnsc CONFIG REQUIRED COMPONENTS plugin_abi plugin_sdk plugin_sdk_crypto
        OPTIONAL_COMPONENTS plugin_sdk_xml)
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()

add_library(example_c_abi MODULE c_abi_driver.c)
target_link_libraries(example_c_abi PRIVATE yaddnsc::plugin_abi)
set(consumer_plugins example_c_abi)

if(NOT YADDNSC_CONSUMER_C_ONLY)
    add_library(example_http MODULE http_driver.cpp)
    target_link_libraries(example_http PRIVATE yaddnsc::plugin_sdk)
    add_library(example_async MODULE async_driver.cpp)
    target_link_libraries(example_async PRIVATE yaddnsc::plugin_sdk)
    add_library(example_crypto MODULE crypto_driver.cpp crypto_hash.cpp)
    target_link_libraries(example_crypto PRIVATE yaddnsc::plugin_sdk yaddnsc::plugin_sdk_crypto)
    if(YADDNSC_CONSUMER_OLD_CXX_ABI)
        target_compile_definitions(example_crypto PRIVATE _GLIBCXX_USE_CXX11_ABI=0)
    endif()
    list(APPEND consumer_plugins example_http example_async example_crypto)
    if(yaddnsc_plugin_sdk_xml_FOUND)
        add_library(example_xml MODULE xml_driver.cpp)
        target_link_libraries(example_xml PRIVATE yaddnsc::plugin_sdk_xml)
        list(APPEND consumer_plugins example_xml)
    endif()
    add_executable(consumer_load consumer_load.cpp)
    target_link_libraries(consumer_load PRIVATE yaddnsc::plugin_abi ${CMAKE_DL_LIBS})
    foreach(plugin IN LISTS consumer_plugins)
        add_test(NAME load_${plugin} COMMAND consumer_load $<TARGET_FILE:${plugin}>)
    endforeach()
endif()

foreach(plugin IN LISTS consumer_plugins)
    string(REPLACE "example_" "consumer_" output_name "${plugin}")
    set_target_properties(${plugin} PROPERTIES OUTPUT_NAME "${output_name}" PREFIX "" C_VISIBILITY_PRESET hidden
        CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endforeach()

# Consumers own their warning policy. Verify the package does not impose one.
foreach(target yaddnsc::plugin_abi yaddnsc::plugin_sdk yaddnsc::plugin_sdk_xml yaddnsc::plugin_sdk_crypto)
    if(TARGET ${target})
        get_target_property(options ${target} INTERFACE_COMPILE_OPTIONS)
        if(options MATCHES "-W(error|all|extra|pedantic)|-fvisibility")
            message(FATAL_ERROR "SDK imposes consumer compile policy: ${options}")
        endif()
    endif()
endforeach()
